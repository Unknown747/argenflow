### 3. TAMBAHKAN FUNGSI DI `bot_engine.py`

#### a) Safety Validator
```python
def cek_keamanan_modal_kecil(self):
    """Ceck apakah modal kecil dan ubah parameter otomatis"""
    if self.saldo < 15:
        self.mode_modal_kecil = True
        self.risiko_pct = RISIKO_PERSEN_MODAL_KECIL
        self.lot = LOT_MAX_MODAL_KECIL
        self.max_positions = MAX_POSISI_MODAL_KECIL
        self.log(f"[SAFETY] Mode Modal Kecil AKTIF | Saldo: ${self.saldo}")
        return True
    return False
```

#### b) Perhitungan Lot Berbasis Risiko
```python
def hitung_lot(self, sl_poin):
    """Hitung lot berdasarkan risk % equity"""
    if self.mode_modal_kecil:
        return self.lot  # fixed lot untuk modal kecil
    
    # Risk-based lot sizing
    risiko_usd = self.saldo * (self.risiko_pct / 100)
    nilai_per_poin = self.hitung_nilai_per_poin()
    lot = risiko_usd / (sl_poin * nilai_per_poin)
    return round(min(lot, self.lot_maks), 2)
```

#### c) Filter Spread & Waktu
```python
def cek_filter(self, symbol):
    spread = mt5.symbol_info(symbol).spread
    if spread > MAX_SPREAD_POIN:
        self.log(f"[SKIP] Spread {spread} poin > {MAX_SPREAD_POIN} (skip)")
        return False
    
    if self.ada_berita_high_impact(HINDARI_NEWS_MENIT):
        self.log(f"[SKIP] Ada berita tinggi dalam {HINDARI_NEWS_MENIT} menit")
        return False
    
    return True
```

#### d) Trailing Stop Otomatis
```python
def update_trailing_stop(self, posisi):
    profit_poin = (posisi.price_current - posisi.price_open) / self.poin_value
    if profit_poin >= MIN_PROFIT_POIN:
        if not posisi.trailing_aktif:
            posisi.trailing_aktif = True
            posisi.sl_trailing = profit_poin - TRAILING_STOP_POIN
        else:
            new_sl = profit_poin - TRAILING_STOP_POIN
            if new_sl > posisi.sl_trailing:
                self.modifikasi_sl(posisi, new_sl)
```
