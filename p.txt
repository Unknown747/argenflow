### 15. DYNAMIC STOP LOSS (ATR-BASED DENGAN BATAS)

```python
# Gunakan ATR tapi dengan batasan aman untuk modal kecil
USE_ATR_SL = True
ATR_PERIOD = 14  # Periode ATR
ATR_MULTIPLIER_SL = 1.0  # 1x ATR untuk SL
ATR_MULTIPLIER_TP = 1.5  # 1.5x ATR untuk TP

# Batasan mutlak (override ATR jika terlalu lebar)
MAX_ATR_SL_POIN = 30  # Stop loss tidak boleh >30 poin meskipun ATR besar
MIN_ATR_SL_POIN = 10  # Stop loss tidak boleh <10 poin (hindari terlalu ketat)

# Fungsi ATR-Dynamic SL
def hitung_sl_atr(self, symbol):
    atr = self.hitung_atr(symbol, ATR_PERIOD)
    sl_poin = atr * ATR_MULTIPLIER_SL
    # Clamp ke batasan
    sl_poin = max(MIN_ATR_SL_POIN, min(MAX_ATR_SL_POIN, sl_poin))
    return sl_poin
```

### KODE `keep_alive.py` LENGKAP

```python
import requests
import time
import threading
import os

URL = os.getenv("REPL_URL", "http://localhost:5000")
PING_INTERVAL = 180  # 3 menit

def keep_alive():
    """Ping endpoint /api/status agar Replit tidak sleep"""
    while True:
        try:
            response = requests.get(f"{URL}/api/status", timeout=10)
            if response.status_code == 200:
                print(f"[KEEPALIVE] Ping success at {time.ctime()}")
            else:
                print(f"[KEEPALIVE] Ping failed: {response.status_code}")
        except Exception as e:
            print(f"[KEEPALIVE] Error: {e}")
        time.sleep(PING_INTERVAL)

# Jalankan di thread terpisah
thread = threading.Thread(target=keep_alive, daemon=True)
thread.start()
```

### 20. AUTO-COMPOUNDING (OPSIONAL UNTUK MODAL KECIL)

```python
# Compounding setting (untuk tumbuh dari $10 ke $50+)
COMPOUND_MODE = True  # Aktifkan compounding
COMPOUND_THRESHOLD = 20  # Mulai compounding setelah profit $20
COMPOUND_RISK_INCREMENT = 0.25  # Tambah risk 0.25% setiap +$10 profit

# Contoh: modal $10 risk 0.5%
# Profit +$20 -> total $30 -> risk jadi 0.75%
# Profit +$20 lagi -> total $50 -> risk jadi 1.0%

# Fungsi auto-compounding
def update_risk_based_on_equity(self):
    if not COMPOUND_MODE:
        return
    profit_above_threshold = max(0, self.saldo - COMPOUND_THRESHOLD)
    level = profit_above_threshold // 10  # setiap $10 profit
    self.risiko_pct = RISIKO_PERSEN_MODAL_KECIL + (level * COMPOUND_RISK_INCREMENT)
    self.risiko_pct = min(self.risiko_pct, 2.0)  # Maksimal 2% risk
```

        ("Belum mencapai target harian", self.daily_profit_target <= self.target_profit_harian),
    ]

            self.log(f"[REJECT] {check_name} failed")
            return False, check_name
